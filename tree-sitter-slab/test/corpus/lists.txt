  col {
    text label
    each children
  }
}
params {
  roots list(Tree) = [
    Tree(
      label="root",
      children=[
        Tree(label="child", children=[]),
      ],
    ),
  ]
}
col scroll {
  each param.roots #tree virtual item-extent=20 overscan=2
}
